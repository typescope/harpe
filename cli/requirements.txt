# Python runtime dependencies for the Harpe loop.
# readline, threading, json, etc. are all in the Python standard library.
#
# Pinned to tested major ranges: the error classification reads SDK exception
# status codes / class names, and client kwargs (timeout, max_retries) must
# exist — both can shift across SDK major versions.
anthropic>=0.40,<1   # required for PROVIDER=anthropic (default)
openai>=2.21,<3      # required for PROVIDER=openai

# Media capabilities (harpe.media). markitdown renders documents (Word, etc.) to
# text — its [all] extra pulls in the image / Office / audio converters. pypdf gives
# page-level PDF access (page count, per-page text).
markitdown[all]>=0.1,<1
pypdf>=6,<7

# Image OCR (harpe.media Image capability): Pillow gives dimensions (it also comes via
# markitdown[all]); pytesseract reads text. pytesseract wraps the Tesseract engine,
# which must be installed as a SYSTEM package (e.g. `apt install tesseract-ocr`) — it
# is NOT a pip dependency. Absent the binary, Image.text returns a message.
pytesseract>=0.3,<1
